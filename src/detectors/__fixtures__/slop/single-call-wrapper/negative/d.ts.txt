import { fetchUser } from "./db";
export function getUserName(id: string) {
  const u = fetchUser(id);
  return u.name.trim();
}
export function getActiveUser(id: string) {
  return fetchUser(id, { activeOnly: true });
}
const ADMIN_ROLES = new Set<string>();
export function isAdminRole(role: string): boolean {
  return ADMIN_ROLES.has(role);
}
