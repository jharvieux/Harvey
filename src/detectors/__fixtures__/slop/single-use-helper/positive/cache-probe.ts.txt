import { fileIsFresh } from "./fs-age.js";

function cacheIsWarm(root: string): boolean {
  return fileIsFresh(`${root}/manifest.json`);
}

export async function warmCache(root: string, fetchAll: () => Promise<void>): Promise<void> {
  if (cacheIsWarm(root)) return;
  await fetchAll();
}
