import { composeStarterResult } from "./starter-compose";

function detectIntent(brief: string): string {
  const normalized = brief.toLowerCase().trim();
  return normalized.includes("migrate") ? "migration" : "prompt";
}

export async function resolveStarter(brief: string, recipe: string) {
  return composeStarterResult({ intent: detectIntent(brief), recipe });
}
