import { QueueWorker } from "job-queue";

function parseQueues(spec: string): { name: string; concurrency: number }[] {
  return spec.split(",").map((part) => {
    const [name, concurrency] = part.split(":");
    return { name: name ?? "default", concurrency: Number(concurrency ?? 1) };
  });
}

export async function startWorkers(spec: string, run: (job: string) => Promise<void>) {
  const queues = parseQueues(spec);
  const workers = queues.map(
    (q) =>
      new QueueWorker(q.name, async (job: string) => {
        await run(job);
      }),
  );
  return { queues, workers };
}
