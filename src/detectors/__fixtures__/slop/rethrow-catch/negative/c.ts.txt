export async function save(): Promise<void> {
  try {
    await write();
  } catch (err) {
    logError(err);
    throw new SaveError("write failed", { cause: err });
  }
}
export async function relay(fallbackErr: Error): Promise<void> {
  try {
    await write();
  } catch (caught) {
    throw fallbackErr;
  }
}
