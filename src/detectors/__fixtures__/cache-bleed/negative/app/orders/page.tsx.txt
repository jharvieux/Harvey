import { unstable_cache } from "next/cache";
import { getServerSession } from "next-auth";
import { supabase } from "../../lib/db";

// Correct per-user caching: the identity is in BOTH the key parts and the tag, so each user
// gets their own Data Cache entry.
const ordersFor = (userId: string) =>
  unstable_cache(
    async (user: string) => {
      const { data } = await supabase.from("orders").select("id, total").eq("user_id", user);
      return data;
    },
    ["orders", userId],
    { revalidate: 3600, tags: [`orders-${userId}`] },
  )(userId);

// Correct `"use cache"` shape: the identity arrives as an ARGUMENT, which is what the cached
// scope is keyed on — no session read happens inside the cached function.
async function orderCountFor(userId: string) {
  "use cache";
  const { count } = await supabase.from("orders").select("id", { count: "exact" }).eq("user_id", userId);
  return count;
}

export default async function OrdersPage() {
  const session = await getServerSession();
  const orders = await ordersFor(session!.user.id);
  const total = await orderCountFor(session!.user.id);

  return (
    <section>
      <p>{total}</p>
      <ul>
        {orders?.map((order) => (
          <li key={order.id}>{order.total}</li>
        ))}
      </ul>
    </section>
  );
}
