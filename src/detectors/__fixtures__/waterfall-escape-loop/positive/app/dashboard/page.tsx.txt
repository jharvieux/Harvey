// #1438 — the true-positive control for the LOOP half of the escape rule (#1484 split this out of
// a bundled fixture that used to hold both the switch and loop shapes together). The intervening
// statement reads the first result and contains a `break`, but the break belongs to the inner
// `for` loop it sits in, not to the function — control reaches the second query either way, so the
// pair is a genuine waterfall and `Promise.all` is safe.
import { supabase } from "../../lib/db";

export async function innerLoopBreakBetweenQueries() {
  const { data: members } = await supabase.from("members").select("id, active");

  for (const m of members ?? []) {
    if (m.active) {
      console.log("first active", m.id);
      break;
    }
  }

  const { data: invoices } = await supabase.from("invoices").select("id, total");
  return { members, invoices };
}
