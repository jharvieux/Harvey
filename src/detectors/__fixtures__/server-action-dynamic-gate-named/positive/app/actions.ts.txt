// Resolving a dynamic import must not turn "this action awaits SOMETHING it imported dynamically"
// into "this action is gated" — the same trap #1263's positive guards for static imports.
// Regressing it means the `named` branch of collectDynamicImports (or GateResolver's identifier
// arm) starts vouching for a resolved callee it never inspected.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";

const RenameSchema = z.object({ orgId: z.string().uuid(), name: z.string().min(1) });

export async function renameOrganisation(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  const { normaliseOrgId } = await import("../lib/format");
  await supabase.from("organisations").update({ name }).eq("id", normaliseOrgId(orgId));
}
