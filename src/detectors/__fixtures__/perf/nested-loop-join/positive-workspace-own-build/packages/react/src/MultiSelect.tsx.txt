interface Option {
  value: string;
  label: string;
}

// The real shape this fixture reproduces: carbon's packages/react/src/MultiSelect.tsx renders a
// selected-value badge list, looking each item up in the full options array on every render.
export function SelectedBadges({ value, options }: { value: string[]; options: Option[] }) {
  return value.map((item) => options.find((option) => option.value === item)?.label);
}
