import { supabase } from "../../../../lib/db";
import { NextResponse } from "next/server";

// An entirely ordinary route name that happens to collide with DEV_TOOLING_PATH's
// `migrations?/` segment. The N+1 below is on the live request path.
export async function POST(req: Request) {
  const { tenantIds } = (await req.json()) as { tenantIds: string[] };
  const applied = [];
  for (const id of tenantIds) {
    const { data } = await supabase.from("schema_migrations").select("id, version").eq("tenant_id", id).limit(1);
    applied.push(data);
  }
  return NextResponse.json(applied);
}
