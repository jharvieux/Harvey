import { supabase } from "../../../../lib/db";
import { NextResponse } from "next/server";

// The scope control: a genuine request-path N+1 under the same tooling-shaped segment.
// It MUST fire, so a zero on this fixture is distinguishable from the scanner never reading it.
export async function POST(req: Request) {
  const { tenantIds } = (await req.json()) as { tenantIds: string[] };
  const applied = [];
  for (const id of tenantIds) {
    const { data } = await supabase.from("schema_migrations").select("id, version").eq("tenant_id", id).limit(1);
    applied.push(data);
  }
  return NextResponse.json(applied);
}
