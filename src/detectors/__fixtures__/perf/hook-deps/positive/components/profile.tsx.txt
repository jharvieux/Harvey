"use client";
import { useEffect, useState } from "react";

export function Profile({ userId }: { userId: string }) {
  const [profile, setProfile] = useState<unknown>(null);
  useEffect(() => {
    fetch(`/api/users/${userId}`)
      .then((r) => r.json())
      .then(setProfile);
  }, []);
  return <pre>{JSON.stringify(profile)}</pre>;
}
