"use server";

// #1434 — the boundary for negative-helper-gate, and deliberately the HARD one: this action's
// only helper also reads the session, so a rule that merely re-tested AUTH_PATTERN against the
// callee's body would suppress this too. `auditLog` writes a log line and returns nothing it can
// deny with, and the action does not consume its result — it enforces nothing (#1439's a1 shape).
//
// The finding must still fire, and its evidence must not claim a helper was checked and cleared.
import { createServiceClient } from "../lib/supabase";
import { auditLog } from "../lib/audit";

export async function updateUserProfile(userId: string, displayName: string) {
  await auditLog("profile.rename", userId);

  const supabaseService = createServiceClient();
  await supabaseService.from("users").update({ display_name: displayName }).eq("id", userId);
}

// The second #1434 shape, and the one that was a FALSE NEGATIVE rather than false evidence: this
// detector tested AUTH_PATTERN against the action's RAW text, so the word in a comment vouched for
// the action and silenced it. #845 blanked literals and comments for the sibling detector and this
// one never got it. Nothing here authenticates anybody.
export async function updateUserEmail(userId: string, email: string) {
  // TODO: requireUser() before shipping
  const supabaseService = createServiceClient();
  await supabaseService.from("users").update({ email }).eq("id", userId);
}
