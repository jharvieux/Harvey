// #1438 — the true-positive controls for the escape rule. Both intervening statements READ the
// first result and contain a `break`, but neither break leaves the function: one belongs to the
// `switch` it sits in, the other to an inner `for`. Control flow reaches the second query either
// way, so both pairs are genuine waterfalls and `Promise.all` is safe on both.
import { supabase } from "../../lib/db";

export async function switchBreakBetweenQueries() {
  const { data: rows } = await supabase.from("rows").select("id, status");

  switch (rows?.[0]?.status) {
    case "open":
      console.log("open");
      break;
    default:
      break;
  }

  const { data: projects } = await supabase.from("projects").select("id, name");
  return { rows, projects };
}

export async function innerLoopBreakBetweenQueries() {
  const { data: members } = await supabase.from("members").select("id, active");

  for (const m of members ?? []) {
    if (m.active) {
      console.log("first active", m.id);
      break;
    }
  }

  const { data: invoices } = await supabase.from("invoices").select("id, total");
  return { members, invoices };
}
