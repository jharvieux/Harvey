// The precision boundary for #1438: the escape rule was narrowed for `break`/`continue`, NOT
// switched off. A `return` inside a `switch` case still leaves the function, so the second query
// does not run on that path and the pair stays a dependency — nothing must fire.
import { supabase } from "../../lib/db";

export async function switchReturnBetweenQueries() {
  const { data: rows } = await supabase.from("rows").select("id, status");

  switch (rows?.[0]?.status) {
    case "archived":
      return null;
    default:
      break;
  }

  const { data: projects } = await supabase.from("projects").select("id, name");
  return { rows, projects };
}
