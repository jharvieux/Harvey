import { UserCard } from "./UserCard";
import { supabase } from "../../lib/db";

// The query ALREADY projected: `select("id, name")` names its columns, so "every field on the row

// carbon's server→client-leak Highs were this shape).
export default async function DashboardPage() {
  const { data: user } = await supabase.from("users").select("id, name").single();

  return <UserCard user={user} />;
}
