import assert from "node:assert";
import { beforeEach, describe, expect, it } from "vitest";
import { buildReport, exportPdf, resetReportCache } from "./report.js";
import { reportChecker } from "./report-checker.js";

function checkReportShape(report: unknown): void {
  expect(report).toMatchObject({ quarter: expect.any(String) });
}

beforeEach(() => {
  resetReportCache();
});

describe("report", () => {
  it("builds the quarterly report", () => {
    expect(buildReport("2026-Q2").rows.length).toBe(12);
  });

  it("exports to pdf", async () => {
    assert.equal(await exportPdf("2026-Q2"), true);
  });

  it("delegates its assertions to a local helper", () => {
    checkReportShape(buildReport("2026-Q2"));
  });

  it("delegates its assertions to an object helper method (page-object shape)", () => {
    reportChecker.checkTotals(buildReport("2026-Q2"));
  });

  it.todo("handles an empty quarter");
});
