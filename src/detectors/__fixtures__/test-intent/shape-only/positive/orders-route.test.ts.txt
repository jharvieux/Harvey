import { describe, expect, it } from "vitest";
import { GET } from "./route";
import { api } from "./client";

function makeRequest(url: string, userId: string) {
  return new Request(url, { headers: { "x-user-id": userId } });
}

describe("GET /api/orders", () => {
  // Parses the body, then asserts only status + count + existence — "returns 200 with 2 rows"
  // passes even if both rows belong to another tenant.
  it("returns the caller's orders", async () => {
    const res = await GET(makeRequest("http://test/api/orders", "user-1"));
    const rows = await res.json();
    expect(res.status).toBe(200);
    expect(rows).toHaveLength(2);
    expect(rows[0].id).toBeDefined();
  });

  // supertest-style: res.body read alongside res.statusCode, only shape/count asserted.
  it("lists team members", async () => {
    const res = await api.get("/api/members");
    expect(res.statusCode).toBe(200);
    expect(res.body.length).toBe(3);
  });
});
