// #1439 — two helpers that MATCH the auth pattern and enforce nothing. #1263 taught the gate check
// to resolve a callee and re-test the pattern against its body; "the body mentions the session" is
// not "the body can stop this mutation". Both actions must still be flagged.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";
import { auditLog } from "../lib/audit";
import { canAccess } from "../lib/access";

const RenameSchema = z.object({ id: z.string().uuid(), name: z.string().min(1) });

// a1 — the action's only helper is a LOGGER. Its body calls getCurrentUser() for the log line; it
// it enforces nothing, and the action does nothing with what it returns.
export async function renameLogged(input: unknown) {
  const { id, name } = RenameSchema.parse(input);
  await auditLog("rename", id);

  await supabase.from("projects").update({ name }).eq("id", id);
}

// a2 — a boolean gate whose result is DISCARDED. `canAccess` reads the session and answers
// truthfully; nobody looks at the answer, so the mutation runs either way.
export async function renameUnchecked(input: unknown) {
  const { id, name } = RenameSchema.parse(input);
  const allowed = await canAccess(id);

  await supabase.from("projects").update({ name }).eq("id", id);
}
