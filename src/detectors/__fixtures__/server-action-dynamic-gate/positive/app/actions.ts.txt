// The adversarial control for #1462. Resolving a dynamic import must not turn "this action awaits
// SOMETHING it imported dynamically" into "this action is gated" — the same trap #1263's positive
// guards for static imports, and the one a sibling batch fell into by accepting a logger and a
// discarded boolean as gates. FOUR shapes, every one of which must STILL be flagged:
//   renameA — a destructured dynamic import that resolves to a helper performing NO check;
//   renameB — a COMPUTED specifier, so the module goes unidentified;
//   renameC — a specifier naming a package, outside the loaded source set;
//   renameD — the namespace form of the same non-gate module.
// The last three are the "unresolvable specifier" half of the acceptance: this pass may only ever
// suppress, so anything it does not evaluate has to leave the finding standing.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";

const RenameSchema = z.object({ orgId: z.string().uuid(), name: z.string().min(1) });

export async function renameA(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  const { normaliseOrgId } = await import("../lib/format");
  await supabase.from("organisations").update({ name }).eq("id", normaliseOrgId(orgId));
}

export async function renameB(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  const specifier = process.env.GATE_MODULE ?? "../lib/format";
  const { normaliseOrgId } = await import(specifier);
  await supabase.from("organisations").update({ name }).eq("id", normaliseOrgId(orgId));
}

export async function renameC(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  const { normaliseOrgId } = await import("@vendor/org-utils");
  await supabase.from("organisations").update({ name }).eq("id", normaliseOrgId(orgId));
}

export async function renameD(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  const format = await import("../lib/format");
  await supabase.from("organisations").update({ name }).eq("id", format.normaliseOrgId(orgId));
}
