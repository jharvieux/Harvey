// #1462 — the FP shape MEASURED on TanStack/tanstack.com's pin: the gate is real, and the only
// reason #1263's resolver could not see it is that `requireAdmin` binds its own check through a
// DYNAMIC `await import()`. Two hops of resolution reach `getAuthenticatedUser`, whose body reads
// the session. Correctly authorized — nothing must fire.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";
import { requireAdmin } from "../lib/roles";

const RenameSchema = z.object({ orgId: z.string().uuid(), name: z.string().min(1) });

export async function renameOrganisation(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  await requireAdmin();

  await supabase.from("organisations").update({ name }).eq("id", orgId);
}
