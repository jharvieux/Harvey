import { expect, test } from "vitest";
import { prisma } from "./setup";

test("update applies the policy filter", async () => {
  const post = await prisma.post.update({ where: { id: 1 }, data: { title: "updated" } });
  expect(post.title).toBe("updated");
});

test("delete removes the row", async () => {
  await prisma.post.delete({ where: { id: 2 } });
  const found = await prisma.post.findUnique({ where: { id: 2 } });
  expect(found).toBeNull();
});

test("updateMany over a fixed id set", async () => {
  await prisma.post.updateMany({ where: { id: { in: [3, 4] } }, data: { published: true } });
});
