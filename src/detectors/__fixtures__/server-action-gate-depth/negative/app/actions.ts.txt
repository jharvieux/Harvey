"use server";

import { z } from "zod";
import { supabase } from "../lib/db";
import { requireModerate } from "../lib/gate";

const ModerateSchema = z.object({ showcaseId: z.string().uuid(), action: z.enum(["approve", "deny"]) });

export async function moderateShowcase(input: unknown) {
  const { showcaseId, action } = ModerateSchema.parse(input);
  await requireModerate();

  await supabase.from("showcases").update({ status: action }).eq("id", showcaseId);
}
