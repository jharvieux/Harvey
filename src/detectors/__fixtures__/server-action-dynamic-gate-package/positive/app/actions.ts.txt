// STRENGTHENED the same way its computed sibling is: `../lib/roles` really does export a real gate

// repo. So the only reason this fires is that the specifier resolves to nothing.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";

const RenameSchema = z.object({ orgId: z.string().uuid(), name: z.string().min(1) });

export async function renameOrganisation(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  const { requireAdmin } = await import("@vendor/org-utils");
  await requireAdmin();

  await supabase.from("organisations").update({ name }).eq("id", orgId);
}
