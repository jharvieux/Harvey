// #1439 — the DISCARDED-BOOLEAN half (#1484 split this out of a bundled fixture that used to hold
// both the logger and discarded-boolean shapes together). `canAccess` reads the session and
// answers truthfully, matching the auth pattern — but nobody looks at the answer, so the mutation
// runs either way. Must still be flagged.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";
import { canAccess } from "../lib/access";

const RenameSchema = z.object({ id: z.string().uuid(), name: z.string().min(1) });

export async function renameUnchecked(input: unknown) {
  const { id, name } = RenameSchema.parse(input);
  const allowed = await canAccess(id);

  await supabase.from("projects").update({ name }).eq("id", id);
}
