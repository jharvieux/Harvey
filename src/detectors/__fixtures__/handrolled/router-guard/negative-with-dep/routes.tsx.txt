// Gate open (react-router is a dep), but neither shape is a guard fork: a redirect-only page is
// the legit use of <Navigate> (no children), and a plain layout renders <Outlet/> with no redirect.
import { Navigate, Outlet } from "react-router-dom";

export function LoginRedirect() {
  return <Navigate to="/login" replace />;
}

export function AppLayout() {
  return (
    <div className="app">
      <Outlet />
    </div>
  );
}
