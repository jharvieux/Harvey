// #1439 — #1263's original false positive, reached through the one import idiom its callee
// resolution did not model. `guards.ensureMember(orgId)` is the real gate. Nothing must fire.
"use server";

import { z } from "zod";
import { supabase } from "../lib/db";
import * as guards from "../lib/guards";

const RenameSchema = z.object({ orgId: z.string().uuid(), name: z.string().min(1) });

export async function renameOrganisation(input: unknown) {
  const { orgId, name } = RenameSchema.parse(input);
  await guards.ensureMember(orgId);

  await supabase.from("organisations").update({ name }).eq("id", orgId);
}
