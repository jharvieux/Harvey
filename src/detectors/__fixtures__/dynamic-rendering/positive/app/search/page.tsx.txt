import { supabase } from "../../lib/db";

export default async function SearchPage({
  searchParams,
}: {
  searchParams: { q?: string };
}) {
  const { data: results } = await supabase
    .from("listings")
    .select("id, title")
    .ilike("title", `%${searchParams.q ?? ""}%`);

  return (
    <ul>
      {results?.map((r) => (
        <li key={r.id}>{r.title}</li>
      ))}
    </ul>
  );
}
